# ======================================================================
# AI Crop Health - server dependencies
# ======================================================================
# Python 3.11
#   pip install -r requirements.txt
#
# This is the SERVER set: everything needed to run the site in production.
# It deliberately leaves out TensorFlow and the training-only libraries -
# see requirements-ml.txt for those and for why.
#
# This file previously contained three concatenated copies of itself with
# conflicting pins (Django>=4.2,<5.0 alongside django==4.2.17, both psycopg
# and psycopg2-binary, tensorflow pinned at two different floors). Rewritten
# as a single list, with upper bounds so a new major release of a dependency
# cannot silently break a deploy.
# ======================================================================

# ---------------------------------------------------------------- Framework
Django>=4.2.17,<5.0
python-dotenv>=1.0.1,<2.0

# ---------------------------------------------------------------- Imaging
Pillow>=11.0,<12.0
# headless, not opencv-python: the server has no display, and the GUI build
# drags in X11/GTK shared libraries that are not present on a slim Linux image
# and add ~300MB for nothing. Same cv2 API.
opencv-python-headless>=4.8,<5.0

# ---------------------------------------------------------------- Classic ML (runtime)
# Used by the crop and fertilizer recommenders, which load small .pkl models.
# Upper bounds matter here specifically: the crop and fertilizer models are
# pickled estimators. A major scikit-learn release changes the internal class
# layout and the .pkl files stop unpickling - on the server, at the moment a
# farmer uses the feature. Verified working on scikit-learn 1.9 / numpy 2.4.
numpy>=1.26,<3.0
scikit-learn>=1.3,<2.0
joblib>=1.3,<2.0

# ---------------------------------------------------------------- External services
google-genai>=0.8.0,<1.0      # Gemini: chatbot fallback provider
requests>=2.32.3,<3.0         # weather, crop.health, insect.id, Groq, translation
markdown>=3.5,<4.0            # Ask AI / chatbot answers are authored in markdown
bleach>=6.0,<7.0              # ...and sanitised before they reach a page

# ---------------------------------------------------------------- Production serving
gunicorn>=21.2,<27.0
whitenoise>=6.5,<7.0
psycopg[binary]>=3.1,<4.0     # PostgreSQL driver (Django 4.2 supports psycopg 3)

# ---------------------------------------------------------------- Optional
# xgboost is imported inside a try/except in
# iot_sensor/services/xgboost_predictor.py and falls back to the existing
# scikit-learn model when absent, so it is genuinely optional. Install it only
# if you have trained an XGBoost model to deploy.
# xgboost>=2.0,<4.0
#
# Error tracking. Recommended once real users are on the system; the settings
# module enables it only when SENTRY_DSN is set.
# sentry-sdk[django]>=2.0,<3.0
